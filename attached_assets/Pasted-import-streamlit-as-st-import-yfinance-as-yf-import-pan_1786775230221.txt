# 2. 緩存抓取各國 ETF 與 SPY 績效
@st.cache_data(ttl=3600)
def fetch_global_performance():
    tickers = [info["etf"] for info in GLOBAL_MARKETS.values()]
    tickers.append("SPY") # 美股基準
    
    df = yf.download(tickers, period="1y")["Close"]
    latest = df.iloc[-1]
    
    m1 = (latest / df.iloc[-21] - 1) * 100
    m3 = (latest / df.iloc[-63] - 1) * 100
    y1 = (latest / df.iloc[0] - 1) * 100
    
    rows = []
    for region, info in GLOBAL_MARKETS.items():
        etf = info["etf"]
        rows.append({
            "國家/區域": region,
            "代表 ETF": etf,
            "近1個月 (%)": round(m1[etf], 2),
            "近3個月 (%)": round(m3[etf], 2),
            "近1年 (%)": round(y1[etf], 2)
        })
        
    perf_df = pd.DataFrame(rows)
    spy_m1 = round(m1["SPY"], 2)
    return perf_df, spy_m1

# 3. Streamlit UI 頁面
def render_global_page():
    st.title("🌐 全球市場與美股 ADR 精選標的")
    st.caption("在以美股為主體的配置下，輕鬆透過美股 ADR 與國家 ETF 佈局歐洲、日本、台灣、韓國、中國與香港。")

    with st.spinner("讀取全球市場即時行情..."):
        try:
            perf_df, spy_m1 = fetch_global_performance()
        except Exception as e:
            st.error(f"數據抓取失敗: {e}")
            return

    # 1. 績效排行榜
    timeframe = st.radio("排序時間範圍", ["近1個月 (%)", "近3個月 (%)", "近1年 (%)"], horizontal=True)
    sorted_df = perf_df.sort_values(by=timeframe, ascending=False).reset_index(drop=True)

    fig = px.bar(
        sorted_df,
        x=timeframe,
        y="國家/區域",
        orientation="h",
        color=timeframe,
        color_continuous_scale="Viridis",
        title=f"全球主要市場 ETF 報酬率表現 ({timeframe})"
    )
    fig.update_layout(yaxis={"categoryorder": "total ascending"})
    st.plotly_chart(fig, use_container_width=True)

    st.markdown("---")

    # 2. 各國市場深度探索與標的推薦
    st.subheader("🔍 分區國際市場與精選美股標的 (ADR / ETF)")
    
    selected_region = st.selectbox("選擇要探索的國家/地區", list(GLOBAL_MARKETS.keys()))
    region_info = GLOBAL_MARKETS[selected_region]

    st.info(f"**市場簡介：** {region_info['description']}")

    col_etf, col_stock = st.columns([1, 2])

    with col_etf:
        st.markdown("### 🧺 國家/區域 ETF")
        for etf in region_info["top_etfs"]:
            st.code(etf, language=None)
            
    with col_stock:
        st.markdown("### 📈 美股市場上市之精選個股 (ADR)")
        for item in region_info["top_stocks"]:
            ticker = item["ticker"]
            c1, c2 = st.columns([3, 1])
            with c1:
                st.write(f"**{item['name']} (`{ticker}`)**")
                st.caption(item["desc"])
            with c2:
                if st.button("個股診斷", key=f"global_{ticker}"):
                    st.session_state["target_ticker"] = ticker
                    st.session_state["active_tab"] = "個股診斷"
                    st.rerun()

if __name__ == "__main__":
    render_global_page()
